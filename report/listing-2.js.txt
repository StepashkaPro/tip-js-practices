export function loadTasks(storage, key, fallbackTasks) {
  try {
    const raw = storage.getItem(key);
    if (raw === null)
      return { ok: true, source: "initial", tasks: copy(fallbackTasks) };
    const data = JSON.parse(raw);
    if (data?.version !== STORAGE_VERSION || !isValidTaskList(data.tasks))
      throw new Error("Неверная схема сохранённых данных");
    return { ok: true, source: "storage", tasks: copy(data.tasks) };
  } catch (error) {
    return {
      ok: false,
      source: "fallback",
      tasks: copy(fallbackTasks),
      error: "Не удалось загрузить данные: " + error.message,
    };
  }
}
export function saveTasks(storage, key, tasks) {
  try {
    if (!isValidTaskList(tasks)) throw new Error("Неверный список задач");
    storage.setItem(key, JSON.stringify({ version: STORAGE_VERSION, tasks }));
    return { ok: true };
  } catch (error) {
    return {
      ok: false,
      error: "Не удалось сохранить данные: " + error.message,
    };
  }
}
// main.js: запуск
loaded = loadTasks(window.localStorage, storageKey, initialTasks);

function persistCurrentTasks(successMessage) {
  const saved = saveTasks(window.localStorage, storageKey, currentTasks);
  elements.storageStatus.classList.toggle("is-warning", !saved.ok);
  elements.storageStatus.textContent = saved.ok
    ? "Изменения сохранены в localStorage."
    : saved.error;
  elements.message.textContent = saved.ok
    ? successMessage
    : `${successMessage} ${saved.error}`;
  renderApp();
  return saved;
}
